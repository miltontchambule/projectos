<script>

let indice = 0;

let mensagens = [
"Olá meu amor ❤️<br><br>Tenho algo especial para ti...",
"Desde que entraste na minha vida, tudo ficou mais bonito ✨",
"O teu sorriso ilumina os meus dias 🌹",
"Prometo estar contigo nos momentos bons e difíceis 💍",
"Queres ser a minha eterna Valentine? ❤️"
];

function abrir(){
    document.getElementById("tampa").style.transform="rotateX(180deg)";
    document.getElementById("carta").style.bottom="40px";
    criarCoracoes();
}

function fechar(){
    document.getElementById("tampa").style.transform="rotateX(0deg)";
    document.getElementById("carta").style.bottom="-170px";
}

document.getElementById("carta").addEventListener("click", function(){

    indice++;

    if(indice < mensagens.length-1){
        document.getElementById("texto").innerHTML = mensagens[indice];
    }

    if(indice === mensagens.length-1){
        mostrarFormulario();
    }

});

function mostrarFormulario(){

    document.getElementById("texto").innerHTML = `
    Queres ser a minha eterna Valentine? ❤️
    <div class="formulario">
        <button class="botao-form" id="sim">Sim</button>
        <button class="botao-form" onclick="respostaClaro()">Claro que sim</button>
    </div>
    `;

    let botaoSim = document.getElementById("sim");

    botaoSim.addEventListener("mouseover", function(){
        botaoSim.style.position="absolute";
        botaoSim.style.left=Math.random()*150+"px";
        botaoSim.style.top=Math.random()*80+"px";
    });
}

function respostaClaro(){

    // explosão de corações
    for(let i=0;i<100;i++){
        let coracao = document.createElement("div");
        coracao.className="coracao";
        coracao.innerHTML="❤️";
        coracao.style.left=Math.random()*window.innerWidth+"px";
        coracao.style.bottom="0px";
        coracao.style.fontSize=(Math.random()*30+20)+"px";
        coracao.style.animationDuration=(Math.random()*3+2)+"s";
        document.body.appendChild(coracao);

        setTimeout(()=>{coracao.remove();},4000);
    }

    // mensagem cinema
    let mensagem = document.createElement("div");
    mensagem.className="mensagem-final";
    mensagem.innerHTML="TE AMO ❤️<br><span class='nome-brilho'>MILTON ❤️</span>";

    document.body.appendChild(mensagem);

    setTimeout(()=>{
        mensagem.style.transform="translate(-50%,-50%) scale(1)";
    },100);
}

function criarCoracoes(){
    for(let i=0;i<15;i++){
        let coracao = document.createElement("div");
        coracao.className="coracao";
        coracao.innerHTML="❤️";
        coracao.style.left=Math.random()*300+"px";
        coracao.style.bottom="0px";
        coracao.style.fontSize=(Math.random()*20+15)+"px";
        document.querySelector(".container").appendChild(coracao);
        setTimeout(()=>{coracao.remove();},4000);
    }
}
